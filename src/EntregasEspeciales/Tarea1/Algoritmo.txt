1.- META
Programa que busca poner a pureba botones, headers, frames y paneles de control
usando las prestaciones de la biblioteca perteneciente a java
de nombre javax.swing.

2.-DATOS
    Frame mainFrame;
    Label headerLabel;
    Label statusLabel;
    Panel controlPanel;

3.-PROCESOS

4.-SALIDA
    4.1 prepareGUI() {
         mainFrame = create frame("java Swing Examples")
         set mainFrame size to (400,400);
         set mainFrame in layout (create GridLayout(3, 1));

         ON_EVENT (Close_Window on mainFrame) DO:
         Terminate_Program()

         headerLabel = create Label("", Label.CENTER);
         statusLabel = create Label("",Label.CENTER);
         set statusLabel size to (350,100);
         controlPanel = create Panel();
         controlPanel.setLayout(create FlowLayout());
         add headerLabel to mainFrame;
         add controlPanel to mainFrame;
         add statusLabel to mainFrame;
         set mainFrame visibility to true;
    }

    4.2 showDialogDemo() {
         headerLabel.setText("Control in action: JOptionPane");
         okButton = create Button("OK");
         javaButton = create Button("Yes/No");
         cancelButton = create Button("Yes/No/Cancel");

         ON_EVENT (Click on okButton) DO:
            Show_message("Welcome to TutorialsPoint.com");

         ON_EVENT (Clicked on javaButton) DO {
            int out;

            show_message("Click any button");
            out = receive();
            SWITCH (out):
               CASE YES_OPTION:
                   set statusLabel text to "Yes selected."
               CASE NO_OPTION:
                   set statusLabel text to "No selected."
            END SWITCH
            }

         ON_EVENT (Click on cancelButton) DO {
             int out;

             show_message("Click any button");
             out = receive();
             SWITCH (out):
                 CASE YES_OPTION:
                     set statusLabel text to "Yes selected."
                 CASE NO_OPTION:
                     set statusLabel text to "No selected."
                 CASE CANCEL_OPTION:
                     set statusLabel text to "Cancel"
             END SWITCH

             add okButton to controlPanel;
             add javaButton to controlPanel;
             add cancelButton to controlPanel;
             set mainFrame visibility to true;
         }
    }
    4.3
    4.4 onCancelButtonPressed() {
                String message;
                int out = JOptionPane.showConfirmDialog(mainFrame,
                        "Click any button",
                        "TutorialsPoint.com",
                        JOptionPane.YES_NO_CANCEL_OPTION,
                        JOptionPane.INFORMATION_MESSAGE);

                message = textPerOption(out);
                statusLabel.setText(message);
    }
    4.5 String textperOption(int option) {
        return switch (option) {
            case 0 -> "Yes selected.";
            case 1 -> "No selected.";
            case 3 -> "Cancel";
            default -> "";
    }
