1.- META
    Demostración práctica de interfaces gráficas con Java Swing. El programa
    ilustra el manejo de eventos de usuario (ActionListeners) y la invocación
    de diálogos modales (JOptionPane) con retroalimentación visual.

2.- DATOS
    OBJETO JFrame mainFrame
    OBJETO JLabel headerLabel
    OBJETO JLabel statusLabel
    OBJETO JPanel controlPanel

3.- PROCESOS
    3.1 textPerOption(ENTERO option)
        SWITCH (option):
            CASE 0:
                RETORNAR "Yes selected."
            CASE 1:
                RETORNAR "No selected."
            CASE 3:
                RETORNAR "Cancel"
            DEFAULT:
                RETORNAR ""
        END SWITCH

4.- SALIDAS
    4.1 prepareGUI()
        mainFrame = create JFrame("Java Swing Examples")
        set mainFrame.size to (400, 400)
        set mainFrame.layout to (create GridLayout(3, 1))

        ON_EVENT (windowClosing ON mainFrame) DO:
            Terminate_Program(0)
        END EVENT

        headerLabel = create JLabel("", JLabel.CENTER)
        statusLabel = create JLabel("", JLabel.CENTER)
        set statusLabel.size to (350, 100)

        controlPanel = create JPanel()
        set controlPanel.layout to (create FlowLayout())

        add headerLabel to mainFrame
        add controlPanel to mainFrame
        add statusLabel to mainFrame
        set mainFrame.visibility to TRUE

    4.2 showDialogDemo()
        set headerLabel.text to "Control in action: JOptionPane"

        okButton = create JButton("OK")
        javaButton = create JButton("Yes/No")
        cancelButton = create JButton("Yes/No/Cancel")

        add OkButtonHandler to okButton
        add JavaButtonHandler to javaButton
        add CancelButtonHandler to cancelButton

        add okButton to controlPanel
        add javaButton to controlPanel
        add cancelButton to controlPanel

        set mainFrame.visibility to TRUE

    4.3 executeDialog(CADENA jopTitle, CADENA jopMessage, ENTERO optionType, ENTERO messageType)
        ENTERO out
        out = showConfirmDialog(mainFrame, jopMessage, jopTitle, optionType, messageType)
        set statusLabel.text to (CALL textPerOption(out))

    4.4 OkButtonMessage()
        CALL showMessageDialog(mainFrame, "Welcome to TutorialsPoint.com")

    4.5 JavaButtonMessage()
        CALL executeDialog("Click any button", "TutorialsPoint.com", YES_NO_OPTION, QUESTION_MESSAGE)

    4.6 CancelButtonMessage()
        CALL executeDialog("Click any button", "TutorialsPoint.com", YES_NO_CANCEL_OPTION, INFORMATION_MESSAGE)

5.- NAVEGACION
    5.1 OkButtonHandler (IMPLEMENTA ActionListener)
        ON_EVENT (actionPerformed) DO:
            CALL OkButtonMessage()
        END EVENT

    5.2 JavaButtonHandler (IMPLEMENTA ActionListener)
        ON_EVENT (actionPerformed) DO:
            CALL JavaButtonMessage()
        END EVENT

    5.3 CancelButtonHandler (IMPLEMENTA ActionListener)
        ON_EVENT (actionPerformed) DO:
            CALL CancelButtonMessage()
        END EVENT

    5.4 CONSTRUCTOR SwingControlDemo()
        CALL prepareGUI()

    5.5 main()
        demo = create SwingControlDemo()
        CALL demo.showDialogDemo()
