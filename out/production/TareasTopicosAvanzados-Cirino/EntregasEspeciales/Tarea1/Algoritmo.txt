1.- META
    Demostración práctica de interfaces gráficas con Java Swing. El programa
    ilustra el manejo de eventos de usuario (ActionListeners),y la invocación
    de diálogos modales (JOptionPane) con retroalimentación visual.

2.- DATOS
    JFrame mainFrame;
    JLabel headerLabel;
    JLabel statusLabel;
    JPanel controlPanel;

3.- PROCESOS
    3.1 textPerOption(int option) {
        SWITCH (option):
            CASE 0:
                return "Yes selected."
            CASE 1:
                return "No selected."
            CASE 3:
                return "Cancel"
            DEFAULT:
                return ""
        END SWITCH
    }

4.- SALIDA
    4.1 prepareGUI() {
        mainFrame = create JFrame("Java Swing Examples");
        set mainFrame size to (400, 400);
        set mainFrame layout to (create GridLayout(3, 1));

        ON_EVENT (windowClosing on mainFrame) DO:
            Terminate_Program(0);

        headerLabel = create JLabel("", JLabel.CENTER);
        statusLabel = create JLabel("", JLabel.CENTER);
        set statusLabel size to (350, 100);

        controlPanel = create JPanel();
        set controlPanel layout to (create FlowLayout());

        add headerLabel to mainFrame;
        add controlPanel to mainFrame;
        add statusLabel to mainFrame;
        set mainFrame visibility to true;
    }

    4.2 showDialogDemo() {
        headerLabel.setText("Control in action: JOptionPane");
        okButton = create JButton("OK");
        javaButton = create JButton("Yes/No");
        cancelButton = create JButton("Yes/No/Cancel");

        add OkButtonHandler to okButton;
        add JavaButtonHandler to javaButton;
        add CancelButtonHandler to cancelButton;

        add okButton to controlPanel;
        add javaButton to controlPanel;
        add cancelButton to controlPanel;
        set mainFrame visibility to true;
    }

    4.3 executeDialog(String jopTitle, String jopMessage, int optionType, int messageType) {
        int out;
        out = JOptionPane.showConfirmDialog(mainFrame, jopMessage, jopTitle, optionType, messageType);
        statusLabel.setText(textPerOption(out));
    }

    4.4 OkButtonMessage() {
        JOptionPane.showMessageDialog(mainFrame, "Welcome to TutorialsPoint.com");
    }

    4.5 JavaButtonMessage() {
        executeDialog("Click any button", "TutorialsPoint.com", YES_NO_OPTION, QUESTION_MESSAGE);
    }

    4.6 CancelButtonMessage() {
        executeDialog("Click any button", "TutorialsPoint.com", YES_NO_CANCEL_OPTION, INFORMATION_MESSAGE);
    }

5.- NAVEGACION
    5.1 OkButtonHandler implements ActionListener {
        ON_EVENT (actionPerformed) DO:
            OkButtonMessage();
    }

    5.2 JavaButtonHandler implements ActionListener {
        ON_EVENT (actionPerformed) DO:
            JavaButtonMessage();
    }

    5.3 CancelButtonHandler implements ActionListener {
        ON_EVENT (actionPerformed) DO:
            CancelButtonMessage();
    }

    5.4 SwingControlDemo() {
        prepareGUI();
    }

    5.5 main() {
        demo = create SwingControlDemo();
        demo.goal();
        demo.showDialogDemo();
    }
